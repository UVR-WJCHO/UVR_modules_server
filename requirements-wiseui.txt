# WiseUI server — every entry point except main_metaobjrecon.py
#
#   comm_hub.py                  broker; also in requirements-metaobj.txt (needs only pyzmq)
#   main_all_hl2_receiver.py     HL2 viewer + user state (--hqs)
#   main_devicebridge.py         Galaxy Watch / phone bridge
#   main_adaptiveMM.py           adaptive multimodal metrics over hl2ss
#   main_handtrack.py            hand tracking + gesture
#   main_handtrack_forecast.py   delay-compensated hand tracking
#
# Traced from each entry point's import chain (imports inside functions
# included), not guessed:
#   main_handtrack -> modules_hand (HandTracker_onnx) -> handtracker_onnx
#                  -> modules_gesture -> gestureclassifier
#                  -> modules_obj (YOLO)
#                  -> _utils/visualize
#   main_handtrack_forecast -> handtracker_onnx
#                           -> delay_nowcasting (ONNX runner, polars schema)
#   main_all_hl2_receiver --hqs -> modules_HQS (OpenCV YuNet), modules_HQS_profile
#                               -> devicebridge/vad (webrtcvad)
#   main_devicebridge -> devicebridge/watch (websockets), phone, vad
#   main_adaptiveMM -> adaptivemm, _hl2ss (PyAV)
#   all -> _comm/hub_client (pyzmq), _comm/*_pb2 (protobuf)
#
# Versions are the ones installed in uvr_integ (RTX 5090, CUDA 12.8, Python 3.10).
#
#   conda create -n uvr_wiseui python=3.10 && conda activate uvr_wiseui
#   pip install torch torchvision --index-url https://download.pytorch.org/whl/cu128
#   pip install -r requirements-wiseui.txt
#
# torch is deliberately NOT pinned here: the wheel has to match your CUDA
# toolkit, so install it first from the right index. 2.7.0+cu128 is what this
# was verified on.
